% ==================================================

%   thesisappendix        -> Lampiran A, Lampiran B, ...
%   thesisappendixsection -> A.1, A.2, B.1, ...
% ==================================================

\makeatletter

% --------------------------------------------------
% Entri utama: Lampiran A, Lampiran B, ...
% --------------------------------------------------
%
% indent  = 0em
% numwidth = 6.8em
%
% numwidth ini memberi ruang untuk teks:
% "Lampiran A"
%
% Efeknya:
% - "Lampiran A" tampil di kiri
% - judul lampiran mulai setelahnya
% - jika judul panjang, baris berikutnya akan sejajar
%   dengan awal judul, bukan dengan margin kiri
% --------------------------------------------------

\providecommand*\l@thesisappendix{}
\renewcommand*\l@thesisappendix{%
  \@dottedtocline{1}{0em}{6.8em}%
}

% --------------------------------------------------
% Entri anak: A.1, A.2, B.1, ...
% --------------------------------------------------
%
% indent   = 6.8em
% numwidth = 3.2em
%
% Hasilnya entry anak masuk ke dalam seperti contoh.
% --------------------------------------------------

\providecommand*\l@thesisappendixsection{}
\renewcommand*\l@thesisappendixsection{%
  \@dottedtocline{2}{6.8em}{3.2em}%
}

% ==================================================
% Custom TOC / LOF / LOT
% ==================================================
%
% tocloft dapat mendefinisikan ulang perintah daftar ketika dokumen
% dimulai. Karena itu override dilakukan pada \AtBeginDocument.
% ==================================================

\makeatletter

\AtBeginDocument{%

  % ------------------------------
  % DAFTAR ISI
  % ------------------------------
  \renewcommand{\tableofcontents}{%
    \thispagestyle{plain}%
    \thesisfrontlisttitle{\contentsname}%
    \vspace{\thesisfrontlistgap}%
    \begingroup
      \thesisfrontlistbodysetup
      \@starttoc{toc}%
    \endgroup

    % File .toc dapat mengubah tocdepth ketika memasuki lampiran.
    % \setcounter bersifat global, jadi kembalikan setelah Daftar Isi
    % selesai agar daftar lain dan dokumen utama tidak terpengaruh.
    \setcounter{tocdepth}{3}%
  }%

  % ------------------------------
  % DAFTAR GAMBAR DAN ILUSTRASI
  % ------------------------------
  \renewcommand{\listoffigures}{%
    \thispagestyle{plain}%
    \thesisfrontlisttitle{\listfigurename}%
    \vspace{\thesisfrontlistgap}%
    \begingroup
      \thesisfrontlistbodysetup
      \@starttoc{lof}%
    \endgroup
  }%

  % ------------------------------
  % DAFTAR TABEL
  % ------------------------------
  \renewcommand{\listoftables}{%
    \thispagestyle{plain}%
    \thesisfrontlisttitle{\listtablename}%
    \vspace{\thesisfrontlistgap}%
    \begingroup
      \thesisfrontlistbodysetup
      \@starttoc{lot}%
    \endgroup
  }%
}

\makeatother

% ==================================================
% SETUP LAMPIRAN
% ==================================================

\newcommand{\thesisappendixsetup}{%

  % Simpan \section asli
  \let\thesisoriginalsection\section

  % Ubah mode ke appendix
  \appendix

  % Format nomor section: A.1, A.2, ...
  \renewcommand{\thesection}{%
    \thechapter.\arabic{section}%
  }%

  % --------------------------------------------------
  % DAFTAR ISI
  % --------------------------------------------------
  % Section lampiran tidak tampil di TOC
  \addtocontents{toc}{%
    \protect\setcounter{tocdepth}{0}%
  }%

  % --------------------------------------------------
  % DAFTAR TABEL
  % --------------------------------------------------
  % Tabel lampiran tidak masuk Daftar Tabel
  \captionsetup[table]{list=false}%
  \captionsetup[longtable]{list=false}%

  % --------------------------------------------------
  % SECTION LAMPIRAN -> masuk ke .lap
  % --------------------------------------------------

  \RenewDocumentCommand{\section}{s o m}{%

    % \section*{...}
    \IfBooleanTF{##1}{%
      \thesisoriginalsection*{##3}%
    }{%

      % \section{...}
      \IfNoValueTF{##2}{%
        \thesisoriginalsection{##3}%
        \addcontentsline{lap}{thesisappendixsection}{%
          \protect\numberline{\thesection}##3%
        }%
      }{%

        % \section[short]{long}
        \thesisoriginalsection[##2]{##3}%
        \addcontentsline{lap}{thesisappendixsection}{%
          \protect\numberline{\thesection}##2%
        }%
      }%
    }%
  }%
}


% ==================================================
% JUDUL SETIAP LAMPIRAN
% ==================================================
%
% Penggunaan:
% \thesisappendixpage{Judul Lampiran}{label}
%
% Lampiran A/B/C tidak dimasukkan ke Daftar Isi, tetapi otomatis
% dimasukkan ke Daftar Lampiran (.lap).
% ==================================================

\newcommand{\thesisappendixpage}[2]{%

  \clearpage

  % A -> B -> C -> ...
  \refstepcounter{chapter}%

  % Anchor harus berada pada halaman lampiran yang benar.
  \phantomsection
  \label{#2}%

  % Tulis ke Daftar Lampiran.
  \addcontentsline{lap}{thesisappendix}{%
    \protect\numberline{Lampiran~\thechapter}#1%
  }%

  \thispagestyle{plain}%

  \begingroup
    \setstretch{1}%
    \setlength{\parindent}{0pt}%
    \setlength{\parskip}{0pt}%

    {\noindent
      \normalfont
      \bfseries
      \fontsize{14pt}{16.8pt}\selectfont
      Lampiran \thechapter\space #1\par
    }%
  \endgroup

  \vspace{14.4pt}%
}
