% ==================================================
% Helper:
% Judul Daftar
% ==================================================
%
% 14 pt
% Bold
% Center
% Leading 21 pt = 1.5 spacing
%
% Tidak ada vertical spacing sebelum judul karena
% margin atas halaman sudah diatur 3 cm oleh geometry.
% ==================================================

\newcommand{\thesisfrontlisttitle}[1]{%
  \begingroup

    % Hilangkan pengaruh konfigurasi paragraf global
    \setlength{\parindent}{0pt}%
    \setlength{\parskip}{0pt}%

    % Reset global one-half spacing
    \setstretch{1}%

    \noindent
    {%
      \centering
      \normalfont
      \bfseries
      \fontsize{14pt}{21pt}\selectfont
      #1\par
    }%

  \endgroup
}


% ==================================================
% Helper:
% Isi Daftar
% ==================================================
%
% 12 pt
% Regular
% Single spacing
% Justify
% Before / after = 0 pt
% ==================================================

\newcommand{\thesisfrontlistbodysetup}{%

  % Single spacing
  \setstretch{1}%

  % Font isi
  \normalfont
  \fontsize{12pt}{14.4pt}\selectfont

  % Paragraph spacing
  \setlength{\parindent}{0pt}%
  \setlength{\parskip}{0pt}%

  % Justify
  \justifying
}


% ==================================================
% Jarak Judul -> Isi
% ==================================================
%
% Isi menggunakan:
%
% 12 pt / 14.4 pt single spacing
%
% sehingga:
%
% 2 spasi = 2 x 14.4 pt = 28.8 pt
% ==================================================

\newlength{\thesisfrontlistgap}
\setlength{\thesisfrontlistgap}{28.8pt}


% ==================================================
% Custom TOC / LOF / LOT
% ==================================================
%
% PENTING:
%
% Package tocloft mendefinisikan ulang:
%
% \tableofcontents
% \listoffigures
% \listoftables
%
% ketika \begin{document}.
%
% Karena itu override kita dipasang melalui
% \AtBeginDocument agar tidak ditimpa lagi.
% ==================================================

\makeatletter

\AtBeginDocument{%

  % ==================================================
  % DAFTAR ISI
  % ==================================================

  \renewcommand{\tableofcontents}{%

    \thispagestyle{plain}%

    % Judul
    \thesisfrontlisttitle{\contentsname}%

    % Jarak 2 spasi
    \vspace{\thesisfrontlistgap}%

    % Isi
    \begingroup
      \thesisfrontlistbodysetup
      \@starttoc{toc}%
    \endgroup
  }%


  % ==================================================
  % DAFTAR GAMBAR DAN ILUSTRASI
  % ==================================================

  \renewcommand{\listoffigures}{%

    \thispagestyle{plain}%

    % Judul
    \thesisfrontlisttitle{\listfigurename}%

    % Jarak 2 spasi
    \vspace{\thesisfrontlistgap}%

    % Isi
    \begingroup
      \thesisfrontlistbodysetup
      \@starttoc{lof}%
    \endgroup
  }%


  % ==================================================
  % DAFTAR TABEL
  % ==================================================

  \renewcommand{\listoftables}{%

    \thispagestyle{plain}%

    % Judul
    \thesisfrontlisttitle{\listtablename}%

    % Jarak 2 spasi
    \vspace{\thesisfrontlistgap}%

    % Isi
    \begingroup
      \thesisfrontlistbodysetup
      \@starttoc{lot}%
    \endgroup
  }%

}

\makeatother

% ==================================================
% Jarak standar gambar
% ==================================================
%
% Pedoman:
%
% teks
%   ↓ 1 spasi
% gambar
%   ↓ 1 spasi
% caption
%   ↓ 1 spasi
% teks berikutnya
%
% Caption = 12 pt, single spacing.
% 1 baseline single spacing ≈ 14.4 pt.
% ==================================================

\newlength{\thesisfigurespace}
\setlength{\thesisfigurespace}{14.4pt}


% ==================================================
% Font Caption
% ==================================================
%
% - 12 pt
% - Regular
% - Single spacing
% ==================================================

\DeclareCaptionFont{thesiscaptionfont}{%
  \normalfont
  \fontsize{12pt}{14.4pt}\selectfont
  \setstretch{1}%
}


% ==================================================
% Caption Gambar
% ==================================================
%
% Caption berada di bawah gambar.
%
% skip:
%   gambar -> caption = 1 spasi
%
% belowskip:
%   dibuat 0 karena caption -> paragraf berikutnya
%   akan dikendalikan oleh \intextsep.
% ==================================================

\captionsetup[figure]{
  name=Gambar,
  font=thesiscaptionfont,
  labelfont=normalfont,
  textfont=normalfont,
  format=hang,
  justification=justified,
  singlelinecheck=false,
  labelsep=space,
  position=bottom,
  skip=\thesisfigurespace,
  belowskip=0pt,
  parskip=0pt
}


% ==================================================
% Jarak gambar terhadap teks
% ==================================================
%
% Karena sebagian besar gambar memakai [H],
% package float menggunakan \intextsep untuk
% jarak sebelum dan sesudah gambar.
%
% Jadi:
%
% paragraf -> gambar = 14.4 pt
% caption -> paragraf = 14.4 pt
% ==================================================

\setlength{\intextsep}{\thesisfigurespace}


% Untuk gambar yang menggunakan [htbp]
% dan ditempatkan di atas/bawah halaman.
\setlength{\textfloatsep}{\thesisfigurespace}


% Jika dua float berdekatan.
\setlength{\floatsep}{\thesisfigurespace}


% ==================================================
% Caption Tabel
% ==================================================
%
% Untuk sementara tetap dipisahkan agar pengaturan
% gambar tidak mengacaukan aturan tabel.
% ==================================================

\captionsetup[table]{
  name=Tabel,
  font=thesiscaptionfont,
  labelfont=normalfont,
  textfont=normalfont,
  format=hang,
  justification=justified,
  singlelinecheck=false,
  labelsep=space
}


% ==================================================
% Caption Tabel
% ==================================================

\captionsetup[table]{
  name=Tabel,
  labelsep=space,
  justification=centering,
  singlelinecheck=true
}


% ==================================================
% Caption Gambar
% ==================================================

\captionsetup[figure]{
  name=Gambar,
  labelsep=space,
  justification=centering,
  singlelinecheck=true
}
